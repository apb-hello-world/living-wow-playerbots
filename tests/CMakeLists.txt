enable_testing()
option(LIVING_SANITIZERS "Run native contracts with ASan and UBSan" ON)
find_package(Threads REQUIRED)
foreach(test_name party_catchup_math party_positioning living_activity living_activity_authority living_activity_epoch living_activity_mailbox living_activity_permissions living_activity_scope living_activity_native_context living_activity_requests living_activity_operations living_activity_gameplay living_activity_resources living_activity_claim_consumption living_activity_spell_resources living_activity_acquisition living_activity_commitments living_profession_job living_activity_native_commit living_activity_item_gain living_purchase_budget living_craft_capture living_profession_settlement)
    add_executable(${test_name} native/${test_name}.cpp)
    target_include_directories(${test_name} PRIVATE ../playerbot)
    target_compile_features(${test_name} PRIVATE cxx_std_17)
    target_compile_definitions(${test_name} PRIVATE MANGOSBOT_ONE)
    target_compile_options(${test_name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    if(LIVING_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${test_name} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${test_name} PRIVATE -fsanitize=address,undefined)
    endif()
    add_test(NAME ${test_name} COMMAND ${test_name})
    set_tests_properties(${test_name} PROPERTIES TIMEOUT 30)
endforeach()
